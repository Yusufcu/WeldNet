Quick Start

# Install dependencies
pip install -r requirements.txt

# Train a model
python train.py \
    --train_dir data/train \
    --val_dir data/val \
    --model_type weldnet \
    --epochs 100

# Run inference
python inference.py \
    --model_path outputs/best_model.pth \
    --input path/to/image.jpg \
    --class_names good,crack,porosity,slag,lack_of_fusion,undercut
